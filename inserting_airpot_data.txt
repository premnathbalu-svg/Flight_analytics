# ---------- FETCH AIRPORT DATA ----------
def get_airport_data(iata_code):
    url = f"https://{API_HOST}/airports/iata/{iata_code}"
    response = requests.get(url, headers=HEADERS)
    response.raise_for_status()
    return response.json()

# ---------- STORE AIRPORT DATA ----------
def store_airport_data(data):
    sql = """
    INSERT INTO airport (
        icao_code,
        iata_code,
        name,
        city,
        country,
        continent,
        latitude,
        longitude,
        timezone
    ) VALUES (%s, %s, %s, %s, %s, %s, %s, %s, %s)
    ON DUPLICATE KEY UPDATE
        name = VALUES(name),
        city = VALUES(city),
        country = VALUES(country),
        continent = VALUES(continent),
        latitude = VALUES(latitude),
        longitude = VALUES(longitude),
        timezone = VALUES(timeZone)
    """

    values = (
        data.get("icao"),
        data.get("iata"),
        data.get("fullName"),
        data.get("municipalityName"),
        data.get("country", {}).get("name"),
        data.get("continent", {}).get("name"),
        data.get("location", {}).get("lat"),
        data.get("location", {}).get("lon"),
        data.get("timeZone")
    )

    cursor.execute(sql, values)
    connection.commit()
